% Base figure style shared by every course (course-images skill).
% Canonical copy: the course-images skill's reference/base.tex. Each course keeps a
% byte-identical copy next to its style layer and \input's it from there.
% Palette is Okabe-Ito (colour-blind safe). Colour names are neutral; the course
% style layer gives each one its meaning (COURSE.md "Figures").
\usepackage{amsmath}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric,arrows.meta,positioning,calc,fit,backgrounds}
\usepackage[paperwidth=45cm,paperheight=32cm,margin=1cm]{geometry}% oversized so previews never clip
\pagestyle{empty}
\renewcommand{\familydefault}{\sfdefault}

\definecolor{cBlue}{HTML}{0072B2}
\definecolor{cVermillion}{HTML}{D55E00}
\definecolor{cGreen}{HTML}{009E73}
\definecolor{cOrange}{HTML}{E69F00}
\definecolor{cPurple}{HTML}{CC79A7}
\definecolor{cInk}{HTML}{222222}    % text and neutral lines
\definecolor{cGrey}{HTML}{D9D9D9}   % neutral fill

\tikzset{
  every picture/.style={font=\small, text=cInk},
  title/.style={font=\large\bfseries, align=center},
  note/.style={draw=cInk!40, fill=cInk!4, rounded corners=2pt, font=\footnotesize, align=left, inner sep=5pt},
  callout/.style={font=\footnotesize, align=left, inner sep=4pt, rounded corners=3pt, thick},
  badge/.style={circle, fill=#1, text=white, font=\scriptsize\bfseries, inner sep=1.5pt, minimum size=0.5cm},
  badge/.default=cInk,
  % arrows: flow is neutral; strong=<colour> and dashflow=<colour> carry a meaning
  flow/.style={-{Stealth[scale=1.0]}, thick, draw=cInk!70},
  strong/.style={-{Stealth[scale=1.1]}, very thick, draw=#1},
  strong/.default=cInk,
  dashflow/.style={-{Stealth[scale=0.9]}, thick, dashed, draw=#1},
  dashflow/.default=cInk,
}
